### 1. 实现 Array.prototype.map

**题目：** 不使用原生 `map` 方法，实现一个功能相同的函数。

**思路：**
- 遍历数组，对每个元素应用回调函数
- 收集结果到新数组
- 保持 this 绑定和索引传递

**实现：**

```javascript
Array.prototype.myMap = function(callback, thisArg) {
    // 边界处理
    if (this == null) {
        throw new TypeError('Array.prototype.myMap called on null or undefined');
    }
    if (typeof callback !== 'function') {
        throw new TypeError(callback + ' is not a function');
    }

    const result = [];
    const arr = Object(this);  // 转换为对象
    const len = arr.length >>> 0;  // 转换为无符号32位整数

    for (let i = 0; i < len; i++) {
        // 检查索引是否存在（稀疏数组）
        if (i in arr) {
            result[i] = callback.call(thisArg, arr[i], i, arr);
        }
    }

    return result;
};

// 测试
const nums = [1, 2, 3, 4];
const doubled = nums.myMap(x => x * 2);
console.log(doubled);  // [2, 4, 6, 8]

// 稀疏数组
const sparse = [1, , 3];
console.log(sparse.myMap(x => x * 2));  // [2, <empty>, 6]
```

**关键点：**
1. `>>> 0` 确保长度为非负整数
2. `i in arr` 处理稀疏数组
3. `callback.call(thisArg, ...)` 正确绑定 this

### 2. 实现防抖（Debounce）

**题目：** 实现一个防抖函数，确保函数在停止调用后延迟执行。

**应用场景：** 搜索框输入、窗口 resize

**思路：**
- 每次调用都清除之前的定时器
- 设置新的定时器
- 支持立即执行选项

**实现：**

```javascript
function debounce(func, delay, immediate = false) {
    let timer = null;

    return function(...args) {
        const context = this;

        // 清除之前的定时器
        if (timer) clearTimeout(timer);

        if (immediate && !timer) {
            // 立即执行模式：第一次调用立即执行
            func.apply(context, args);
        }

        timer = setTimeout(() => {
            if (!immediate) {
                func.apply(context, args);
            }
            timer = null;  // 重置定时器
        }, delay);
    };
}

// 测试
const logInput = debounce((value) => {
    console.log('Search:', value);
}, 500);

// 模拟输入
logInput('a');     // 不执行
logInput('ab');    // 不执行
logInput('abc');   // 500ms 后执行：Search: abc

// 支持取消
function debounceWithCancel(func, delay) {
    let timer = null;

    const debounced = function(...args) {
        if (timer) clearTimeout(timer);
        timer = setTimeout(() => {
            func.apply(this, args);
        }, delay);
    };

    debounced.cancel = function() {
        clearTimeout(timer);
        timer = null;
    };

    return debounced;
}

const searchDebounced = debounceWithCancel(query => console.log(query), 500);
searchDebounced('test');
searchDebounced.cancel();  // 取消执行
```

**关键点：**
1. 使用闭包保存 `timer`
2. `apply(context, args)` 保持 this 和参数
3. 支持立即执行和取消功能

### 3. 实现节流（Throttle）

**题目：** 实现一个节流函数，确保函数在指定时间内只执行一次。

**应用场景：** 滚动事件、鼠标移动

**思路：**
- 记录上次执行时间
- 计算时间间隔
- 超过阈值才执行

**实现：**

```javascript
function throttle(func, delay) {
    let lastTime = 0;
    let timer = null;

    return function(...args) {
        const context = this;
        const now = Date.now();

        // 剩余等待时间
        const remaining = delay - (now - lastTime);

        if (remaining <= 0) {
            // 可以立即执行
            if (timer) {
                clearTimeout(timer);
                timer = null;
            }
            lastTime = now;
            func.apply(context, args);
        } else if (!timer) {
            // 设置定时器，确保最后一次调用也能执行
            timer = setTimeout(() => {
                lastTime = Date.now();
                timer = null;
                func.apply(context, args);
            }, remaining);
        }
    };
}

// 测试
let count = 0;
const throttled = throttle(() => {
    count++;
    console.log('Executed:', count);
}, 1000);

// 模拟快速调用
setInterval(throttled, 100);  // 每隔 1 秒才执行一次

// 时间戳版本（简化版）
function throttleSimple(func, delay) {
    let lastTime = 0;

    return function(...args) {
        const now = Date.now();
        if (now - lastTime >= delay) {
            lastTime = now;
            func.apply(this, args);
        }
    };
}
```

**关键点：**
1. 时间戳 vs 定时器：前者首次立即执行，后者确保最后一次执行
2. 结合两者可实现更精确的节流

### 4. 实现深拷贝（Deep Clone）

**题目：** 实现一个深拷贝函数，支持对象、数组、循环引用等。

**思路：**
- 递归复制所有属性
- 使用 WeakMap 处理循环引用
- 处理特殊对象（Date、RegExp、Map、Set）

**实现：**

```javascript
function deepClone(obj, hash = new WeakMap()) {
    // 处理 null 和基本类型
    if (obj === null || typeof obj !== 'object') {
        return obj;
    }

    // 处理日期
    if (obj instanceof Date) {
        return new Date(obj);
    }

    // 处理正则
    if (obj instanceof RegExp) {
        return new RegExp(obj.source, obj.flags);
    }

    // 处理 Map
    if (obj instanceof Map) {
        const cloneMap = new Map();
        obj.forEach((value, key) => {
            cloneMap.set(key, deepClone(value, hash));
        });
        return cloneMap;
    }

    // 处理 Set
    if (obj instanceof Set) {
        const cloneSet = new Set();
        obj.forEach(value => {
            cloneSet.add(deepClone(value, hash));
        });
        return cloneSet;
    }

    // 处理循环引用
    if (hash.has(obj)) {
        return hash.get(obj);
    }

    // 处理数组和普通对象
    const clone = Array.isArray(obj) ? [] : {};
    hash.set(obj, clone);

    // 处理 Symbol 属性
    const symbols = Object.getOwnPropertySymbols(obj);
    symbols.forEach(sym => {
        clone[sym] = deepClone(obj[sym], hash);
    });

    // 复制所有可枚举属性
    for (let key in obj) {
        if (obj.hasOwnProperty(key)) {
            clone[key] = deepClone(obj[key], hash);
        }
    }

    return clone;
}

// 测试
const obj = {
    num: 1,
    str: 'hello',
    arr: [1, 2, {a: 3}],
    date: new Date(),
    reg: /test/gi,
    map: new Map([['key', 'value']]),
    set: new Set([1, 2, 3]),
    [Symbol('id')]: 'symbol-value'
};

// 循环引用
obj.self = obj;

const cloned = deepClone(obj);
console.log(cloned);
console.log(cloned.self === cloned);  // true
console.log(cloned !== obj);  // true
```

**关键点：**
1. `WeakMap` 用于存储已复制对象，避免循环引用导致栈溢出
2. 特殊对象需要使用构造函数创建
3. 处理 Symbol 属性

### 5. 实现 Promise.all

**题目：** 手写 `Promise.all` 实现。

**思路：**
- 返回新的 Promise
- 追踪所有 Promise 的状态
- 任一失败则 reject，全部成功则 resolve

**实现：**

```javascript
function promiseAll(promises) {
    return new Promise((resolve, reject) => {
        // 边界处理
        if (!Array.isArray(promises)) {
            return reject(new TypeError('Argument must be an array'));
        }

        if (promises.length === 0) {
            return resolve([]);
        }

        const results = [];
        let completedCount = 0;

        promises.forEach((promise, index) => {
            // 确保是 Promise（支持非 Promise 值）
            Promise.resolve(promise)
                .then(value => {
                    results[index] = value;  // 保持顺序
                    completedCount++;

                    // 所有 Promise 都成功
                    if (completedCount === promises.length) {
                        resolve(results);
                    }
                })
                .catch(error => {
                    // 任一 Promise 失败，立即 reject
                    reject(error);
                });
        });
    });
}

// 测试
const p1 = Promise.resolve(1);
const p2 = new Promise(resolve => setTimeout(() => resolve(2), 100));
const p3 = Promise.resolve(3);

promiseAll([p1, p2, p3])
    .then(results => console.log(results))  // [1, 2, 3]
    .catch(error => console.error(error));

// 测试失败情况
const p4 = Promise.reject('error');
promiseAll([p1, p4, p3])
    .catch(error => console.error(error));  // 'error'
```

**关键点：**
1. 使用 `results[index]` 而非 `push` 保持顺序
2. `Promise.resolve()` 包装支持非 Promise 值
3. 计数器追踪完成状态

### 6. 实现 Promise.race

**题目：** 手写 `Promise.race` 实现。

**思路：**
- 返回第一个完成（fulfilled 或 rejected）的 Promise 结果

**实现：**

```javascript
function promiseRace(promises) {
    return new Promise((resolve, reject) => {
        if (!Array.isArray(promises)) {
            return reject(new TypeError('Argument must be an array'));
        }

        if (promises.length === 0) {
            return;  // 永远不会 resolve/reject
        }

        promises.forEach(promise => {
            Promise.resolve(promise)
                .then(resolve)  // 第一个成功
                .catch(reject);  // 第一个失败
        });
    });
}

// 测试
const p1 = new Promise(resolve => setTimeout(() => resolve('slow'), 500));
const p2 = new Promise(resolve => setTimeout(() => resolve('fast'), 100));

promiseRace([p1, p2])
    .then(result => console.log(result));  // 'fast'
```

### 7. 实现发布-订阅模式（EventEmitter）

**题目：** 实现一个事件发射器，支持 on、off、emit、once 方法。

**实现：**

```javascript
class EventEmitter {
    constructor() {
        this.events = {};  // {事件名: [回调函数]}
    }

    // 订阅事件
    on(event, callback) {
        if (!this.events[event]) {
            this.events[event] = [];
        }
        this.events[event].push(callback);
        return this;  // 链式调用
    }

    // 取消订阅
    off(event, callback) {
        if (!this.events[event]) return this;

        if (callback) {
            // 移除特定回调
            this.events[event] = this.events[event].filter(cb => cb !== callback);
        } else {
            // 移除所有回调
            delete this.events[event];
        }

        return this;
    }

    // 触发事件
    emit(event, ...args) {
        if (!this.events[event]) return false;

        this.events[event].forEach(callback => {
            callback.apply(this, args);
        });

        return true;
    }

    // 订阅一次
    once(event, callback) {
        const wrapper = (...args) => {
            callback.apply(this, args);
            this.off(event, wrapper);  // 执行后移除
        };

        this.on(event, wrapper);
        return this;
    }
}

// 测试
const emitter = new EventEmitter();

const handler1 = (data) => console.log('Handler 1:', data);
const handler2 = (data) => console.log('Handler 2:', data);

emitter.on('message', handler1);
emitter.on('message', handler2);

emitter.emit('message', 'Hello');
// Handler 1: Hello
// Handler 2: Hello

emitter.off('message', handler1);
emitter.emit('message', 'World');
// Handler 2: World

emitter.once('test', (data) => console.log('Once:', data));
emitter.emit('test', 'A');  // Once: A
emitter.emit('test', 'B');  // 不触发
```

**关键点：**
1. 使用对象存储事件 → 回调数组的映射
2. `once` 通过包装函数实现自动移除
3. 支持链式调用

### 8. 实现函数柯里化（Curry）

**题目：** 实现一个柯里化函数，将多参数函数转换为单参数函数序列。

**思路：**
- 收集参数
- 参数足够时执行原函数
- 参数不足时返回新函数继续收集

**实现：**

```javascript
function curry(func) {
    return function curried(...args) {
        // 参数足够，执行原函数
        if (args.length >= func.length) {
            return func.apply(this, args);
        }

        // 参数不足，返回新函数
        return function(...nextArgs) {
            return curried.apply(this, args.concat(nextArgs));
        };
    };
}

// 测试
function add(a, b, c) {
    return a + b + c;
}

const curriedAdd = curry(add);

console.log(curriedAdd(1)(2)(3));  // 6
console.log(curriedAdd(1, 2)(3));  // 6
console.log(curriedAdd(1)(2, 3));  // 6
console.log(curriedAdd(1, 2, 3));  // 6

// 实现占位符支持
function curryWithPlaceholder(func, placeholder = '_') {
    return function curried(...args) {
        // 检查是否有占位符
        if (args.length >= func.length && !args.includes(placeholder)) {
            return func.apply(this, args);
        }

        return function(...nextArgs) {
            // 替换占位符
            const combinedArgs = args.map(arg =>
                arg === placeholder && nextArgs.length ? nextArgs.shift() : arg
            );
            return curried.apply(this, [...combinedArgs, ...nextArgs]);
        };
    };
}

const curriedWithPlaceholder = curryWithPlaceholder(add);
console.log(curriedWithPlaceholder('_', 2, 3)(1));  // 6
```

**关键点：**
1. `func.length` 获取函数参数个数
2. 递归收集参数
3. 占位符支持更灵活的参数传递

### 9. 实现 instanceof

**题目：** 手写 `instanceof` 运算符的实现。

**思路：**
- 沿着原型链查找
- 检查对象的原型是否在构造函数的原型链上

**实现：**

```javascript
function myInstanceof(obj, constructor) {
    // 边界处理
    if (obj === null || obj === undefined) {
        return false;
    }

    if (typeof constructor !== 'function') {
        throw new TypeError('Right-hand side of instanceof is not a function');
    }

    // 获取对象的原型
    let proto = Object.getPrototypeOf(obj);

    // 获取构造函数的 prototype
    const prototype = constructor.prototype;

    // 沿原型链查找
    while (proto !== null) {
        if (proto === prototype) {
            return true;
        }
        proto = Object.getPrototypeOf(proto);
    }

    return false;
}

// 测试
console.log(myInstanceof([], Array));  // true
console.log(myInstanceof([], Object));  // true
console.log(myInstanceof({}, Array));  // false

function Person() {}
const person = new Person();
console.log(myInstanceof(person, Person));  // true
console.log(myInstanceof(person, Object));  // true
```

**关键点：**
1. `Object.getPrototypeOf()` 获取原型
2. 循环遍历原型链
3. 基本类型返回 false

### 10. 实现 new 运算符

**题目：** 手写 `new` 运算符的实现。

**思路：**
1. 创建空对象
2. 设置原型链
3. 执行构造函数
4. 返回对象

**实现：**

```javascript
function myNew(constructor, ...args) {
    // 1. 创建一个空对象，原型指向构造函数的 prototype
    const obj = Object.create(constructor.prototype);

    // 2. 执行构造函数，绑定 this
    const result = constructor.apply(obj, args);

    // 3. 如果构造函数返回对象，则返回该对象；否则返回创建的对象
    return result instanceof Object ? result : obj;
}

// 测试
function Person(name, age) {
    this.name = name;
    this.age = age;
}

Person.prototype.sayHello = function() {
    console.log(`Hello, I'm ${this.name}`);
};

const person = myNew(Person, 'Alice', 25);
console.log(person.name);  // Alice
person.sayHello();  // Hello, I'm Alice

// 测试构造函数返回对象的情况
function Animal(name) {
    this.name = name;
    return {customName: 'Custom'};  // 返回自定义对象
}

const animal = myNew(Animal, 'Dog');
console.log(animal.customName);  // Custom
console.log(animal.name);  // undefined
```

**关键点：**
1. `Object.create()` 创建对象并设置原型
2. `apply` 绑定 this 并传递参数
3. 处理构造函数返回对象的情况

### 11. 实现 async/await（使用 Generator）

**题目：** 使用 Generator 函数模拟 `async/await` 的实现。

**思路：**
- Generator 函数产生迭代器
- 递归执行 `next()`
- 处理 Promise

**实现：**

```javascript
function asyncToGenerator(generatorFunc) {
    return function(...args) {
        const generator = generatorFunc.apply(this, args);

        return new Promise((resolve, reject) => {
            function step(key, arg) {
                let result;
                try {
                    result = generator[key](arg);
                } catch (error) {
                    return reject(error);
                }

                const {value, done} = result;

                if (done) {
                    return resolve(value);
                } else {
                    // 确保 value 是 Promise
                    return Promise.resolve(value).then(
                        val => step('next', val),
                        err => step('throw', err)
                    );
                }
            }

            step('next');
        });
    };
}

// 测试
function* fetchData() {
    try {
        const user = yield fetch('/api/user').then(res => res.json());
        console.log('User:', user);

        const posts = yield fetch(`/api/posts/${user.id}`).then(res => res.json());
        console.log('Posts:', posts);

        return posts;
    } catch (error) {
        console.error('Error:', error);
    }
}

const asyncFetchData = asyncToGenerator(fetchData);
asyncFetchData().then(data => console.log('Done:', data));

// 等价的 async/await 写法
async function fetchDataAsync() {
    try {
        const user = await fetch('/api/user').then(res => res.json());
        console.log('User:', user);

        const posts = await fetch(`/api/posts/${user.id}`).then(res => res.json());
        console.log('Posts:', posts);

        return posts;
    } catch (error) {
        console.error('Error:', error);
    }
}
```

**关键点：**
1. Generator 的 `next()` 和 `throw()` 方法
2. 递归执行直到 `done` 为 true
3. 错误处理通过 `throw()` 传递

### 12. 实现虚拟 DOM 的 diff 算法（简化版）

**题目：** 实现一个简化的虚拟 DOM diff 算法。

**思路：**
- 比较新旧虚拟 DOM 树
- 生成补丁（patch）
- 应用补丁更新真实 DOM

**实现：**

```javascript
// 虚拟 DOM 节点
class VNode {
    constructor(tag, props, children) {
        this.tag = tag;
        this.props = props || {};
        this.children = children || [];
        this.key = props?.key;
    }
}

// 补丁类型
const PATCH_TYPE = {
    REPLACE: 'REPLACE',  // 替换节点
    PROPS: 'PROPS',      // 更新属性
    TEXT: 'TEXT',        // 文本变化
    REORDER: 'REORDER'   // 子节点重排
};

// diff 算法
function diff(oldVNode, newVNode) {
    // 文本节点
    if (typeof oldVNode === 'string' || typeof newVNode === 'string') {
        if (oldVNode !== newVNode) {
            return {type: PATCH_TYPE.TEXT, content: newVNode};
        }
        return null;
    }

    // 节点类型不同，直接替换
    if (oldVNode.tag !== newVNode.tag) {
        return {type: PATCH_TYPE.REPLACE, vNode: newVNode};
    }

    // 比较属性
    const propsPatches = diffProps(oldVNode.props, newVNode.props);

    // 比较子节点
    const childrenPatches = diffChildren(oldVNode.children, newVNode.children);

    if (propsPatches || childrenPatches.length > 0) {
        return {
            type: PATCH_TYPE.PROPS,
            props: propsPatches,
            children: childrenPatches
        };
    }

    return null;
}

// 比较属性
function diffProps(oldProps, newProps) {
    const patches = {};
    let hasChange = false;

    // 检查新增和修改
    for (let key in newProps) {
        if (oldProps[key] !== newProps[key]) {
            patches[key] = newProps[key];
            hasChange = true;
        }
    }

    // 检查删除
    for (let key in oldProps) {
        if (!(key in newProps)) {
            patches[key] = null;
            hasChange = true;
        }
    }

    return hasChange ? patches : null;
}

// 比较子节点（简化版，不考虑 key）
function diffChildren(oldChildren, newChildren) {
    const patches = [];
    const len = Math.max(oldChildren.length, newChildren.length);

    for (let i = 0; i < len; i++) {
        patches.push(diff(oldChildren[i], newChildren[i]));
    }

    return patches;
}

// 应用补丁
function patch(domNode, patches) {
    if (!patches) return;

    switch (patches.type) {
        case PATCH_TYPE.REPLACE:
            const newNode = render(patches.vNode);
            domNode.parentNode.replaceChild(newNode, domNode);
            break;

        case PATCH_TYPE.TEXT:
            domNode.textContent = patches.content;
            break;

        case PATCH_TYPE.PROPS:
            // 更新属性
            if (patches.props) {
                for (let key in patches.props) {
                    if (patches.props[key] === null) {
                        domNode.removeAttribute(key);
                    } else {
                        domNode.setAttribute(key, patches.props[key]);
                    }
                }
            }

            // 更新子节点
            if (patches.children) {
                patches.children.forEach((childPatch, i) => {
                    patch(domNode.childNodes[i], childPatch);
                });
            }
            break;
    }
}

// 渲染虚拟 DOM 为真实 DOM
function render(vNode) {
    if (typeof vNode === 'string') {
        return document.createTextNode(vNode);
    }

    const el = document.createElement(vNode.tag);

    // 设置属性
    for (let key in vNode.props) {
        el.setAttribute(key, vNode.props[key]);
    }

    // 渲染子节点
    vNode.children.forEach(child => {
        el.appendChild(render(child));
    });

    return el;
}

// 测试
const oldVNode = new VNode('div', {class: 'container'}, [
    new VNode('h1', {}, ['Hello']),
    new VNode('p', {}, ['World'])
]);

const newVNode = new VNode('div', {class: 'container', id: 'app'}, [
    new VNode('h1', {}, ['Hi']),
    new VNode('p', {}, ['World']),
    new VNode('span', {}, ['New'])
]);

const patches = diff(oldVNode, newVNode);
console.log('Patches:', patches);

// 应用补丁
const domNode = render(oldVNode);
document.body.appendChild(domNode);
patch(domNode, patches);
```

**关键点：**
1. 深度优先遍历比较
2. 不同节点类型的处理策略
3. 实际框架（React、Vue）使用更复杂的 key-based diff

### 13. 实现 LazyMan（链式调用 + 任务队列）

**题目：** 实现一个 LazyMan，支持链式调用和延迟执行。

```javascript
// 期望用法
LazyMan('Hank')
  .sleep(3)
  .eat('dinner')
  .sleepFirst(2)
  .eat('breakfast');

// 输出：
// Hi, I'm Hank (立即)
// Wake up after 2 (2秒后)
// Wake up after 3 (再过3秒)
// Eat dinner
// Eat breakfast
```

**实现：**

```javascript
class _LazyMan {
    constructor(name) {
        this.name = name;
        this.tasks = [];

        // 初始任务
        this.tasks.push(() => {
            console.log(`Hi, I'm ${name}`);
            this.next();
        });

        // 异步执行任务队列
        setTimeout(() => {
            this.next();
        }, 0);
    }

    next() {
        const task = this.tasks.shift();
        task && task();
    }

    eat(food) {
        this.tasks.push(() => {
            console.log(`Eat ${food}`);
            this.next();
        });
        return this;  // 链式调用
    }

    sleep(time) {
        this.tasks.push(() => {
            setTimeout(() => {
                console.log(`Wake up after ${time}`);
                this.next();
            }, time * 1000);
        });
        return this;
    }

    sleepFirst(time) {
        // 插入到队列开头
        this.tasks.unshift(() => {
            setTimeout(() => {
                console.log(`Wake up after ${time}`);
                this.next();
            }, time * 1000);
        });
        return this;
    }
}

function LazyMan(name) {
    return new _LazyMan(name);
}

// 测试
LazyMan('Hank')
    .sleep(1)
    .eat('dinner')
    .sleepFirst(2)
    .eat('breakfast');
```

**关键点：**
1. 任务队列存储函数
2. `setTimeout(..., 0)` 确保链式调用完成后才开始执行
3. `sleepFirst` 使用 `unshift` 插入到队列开头
4. 每个任务执行完调用 `next()` 继续

### 14. 实现图片懒加载

**需求：** 当图片进入视口时才加载，提升页面性能。

**实现：**

```javascript
class LazyLoad {
    constructor(selector, options = {}) {
        this.images = document.querySelectorAll(selector);
        this.options = {
            root: null,  // 视口
            rootMargin: '0px',
            threshold: 0.01,  // 1% 可见时触发
            ...options
        };

        this.init();
    }

    init() {
        // 使用 IntersectionObserver API
        if ('IntersectionObserver' in window) {
            this.observer = new IntersectionObserver(
                this.handleIntersect.bind(this),
                this.options
            );

            this.images.forEach(img => {
                this.observer.observe(img);
            });
        } else {
            // 降级方案：监听滚动事件
            this.loadImagesOnScroll();
            window.addEventListener('scroll', this.throttle(this.loadImagesOnScroll.bind(this), 200));
        }
    }

    handleIntersect(entries) {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                this.loadImage(entry.target);
                this.observer.unobserve(entry.target);
            }
        });
    }

    loadImage(img) {
        const src = img.dataset.src;
        if (!src) return;

        // 创建新图片对象预加载
        const image = new Image();
        image.onload = () => {
            img.src = src;
            img.classList.add('loaded');
        };
        image.onerror = () => {
            img.classList.add('error');
        };
        image.src = src;
    }

    loadImagesOnScroll() {
        this.images.forEach(img => {
            if (this.isInViewport(img)) {
                this.loadImage(img);
            }
        });
    }

    isInViewport(element) {
        const rect = element.getBoundingClientRect();
        return (
            rect.top >= 0 &&
            rect.left >= 0 &&
            rect.bottom <= (window.innerHeight || document.documentElement.clientHeight) &&
            rect.right <= (window.innerWidth || document.documentElement.clientWidth)
        );
    }

    throttle(func, delay) {
        let lastTime = 0;
        return function(...args) {
            const now = Date.now();
            if (now - lastTime >= delay) {
                lastTime = now;
                func.apply(this, args);
            }
        };
    }
}

// 使用
// HTML: <img data-src="real-image.jpg" src="placeholder.jpg" alt="Lazy">
const lazyLoad = new LazyLoad('img[data-src]', {
    rootMargin: '50px'  // 提前 50px 加载
});
```

**关键点：**
1. 优先使用 `IntersectionObserver`（性能更好）
2. 降级方案：监听 scroll + 节流
3. 预加载处理 onload/onerror

### 15. 实现无限滚动（Infinite Scroll）

**需求：** 滚动到底部时自动加载更多内容。

**实现：**

```javascript
class InfiniteScroll {
    constructor(options) {
        this.container = options.container || window;
        this.threshold = options.threshold || 100;  // 距离底部阈值
        this.loadMore = options.loadMore;  // 加载函数
        this.loading = false;
        this.hasMore = true;

        this.init();
    }

    init() {
        this.handleScroll = this.throttle(this.checkScroll.bind(this), 200);
        this.container.addEventListener('scroll', this.handleScroll);
    }

    checkScroll() {
        if (this.loading || !this.hasMore) return;

        const scrollTop = this.container === window
            ? window.pageYOffset || document.documentElement.scrollTop
            : this.container.scrollTop;

        const scrollHeight = this.container === window
            ? document.documentElement.scrollHeight
            : this.container.scrollHeight;

        const clientHeight = this.container === window
            ? window.innerHeight
            : this.container.clientHeight;

        // 距离底部的距离
        const distanceToBottom = scrollHeight - scrollTop - clientHeight;

        if (distanceToBottom < this.threshold) {
            this.load();
        }
    }

    async load() {
        this.loading = true;

        try {
            const hasMore = await this.loadMore();
            this.hasMore = hasMore !== false;  // 返回 false 表示没有更多数据
        } catch (error) {
            console.error('Failed to load more:', error);
        } finally {
            this.loading = false;
        }
    }

    throttle(func, delay) {
        let lastTime = 0;
        return function(...args) {
            const now = Date.now();
            if (now - lastTime >= delay) {
                lastTime = now;
                func.apply(this, args);
            }
        };
    }

    destroy() {
        this.container.removeEventListener('scroll', this.handleScroll);
    }
}

// 使用
let page = 1;
const infiniteScroll = new InfiniteScroll({
    container: document.querySelector('.content'),
    threshold: 200,
    loadMore: async () => {
        // 模拟 API 请求
        const response = await fetch(`/api/items?page=${page}`);
        const data = await response.json();

        // 渲染数据
        data.items.forEach(item => {
            const div = document.createElement('div');
            div.textContent = item.title;
            document.querySelector('.content').appendChild(div);
        });

        page++;
        return data.hasMore;  // 返回是否还有更多数据
    }
});
```

**关键点：**
1. 计算距离底部的距离
2. 节流防止频繁触发
3. loading 状态防止重复请求

### 16. 实现拖拽排序

**需求：** 实现一个可拖拽排序的列表。

**实现：**

```javascript
class DragSort {
    constructor(container) {
        this.container = container;
        this.items = Array.from(container.children);
        this.dragging = null;

        this.init();
    }

    init() {
        this.items.forEach(item => {
            item.draggable = true;

            item.addEventListener('dragstart', this.handleDragStart.bind(this));
            item.addEventListener('dragover', this.handleDragOver.bind(this));
            item.addEventListener('drop', this.handleDrop.bind(this));
            item.addEventListener('dragend', this.handleDragEnd.bind(this));
        });
    }

    handleDragStart(e) {
        this.dragging = e.target;
        e.dataTransfer.effectAllowed = 'move';
        e.target.classList.add('dragging');
    }

    handleDragOver(e) {
        e.preventDefault();
        e.dataTransfer.dropEffect = 'move';

        const target = e.target.closest('[draggable="true"]');
        if (!target || target === this.dragging) return;

        // 计算鼠标位置，判断插入位置
        const rect = target.getBoundingClientRect();
        const midpoint = rect.top + rect.height / 2;

        if (e.clientY < midpoint) {
            // 插入到目标元素之前
            this.container.insertBefore(this.dragging, target);
        } else {
            // 插入到目标元素之后
            this.container.insertBefore(this.dragging, target.nextSibling);
        }
    }

    handleDrop(e) {
        e.preventDefault();
    }

    handleDragEnd(e) {
        e.target.classList.remove('dragging');
        this.dragging = null;

        // 更新顺序
        this.items = Array.from(this.container.children);
        this.onSort && this.onSort(this.items.map(item => item.dataset.id));
    }

    onSort(callback) {
        this.onSort = callback;
    }
}

// 使用
// HTML:
// <ul id="list">
//   <li data-id="1" draggable="true">Item 1</li>
//   <li data-id="2" draggable="true">Item 2</li>
//   <li data-id="3" draggable="true">Item 3</li>
// </ul>

const dragSort = new DragSort(document.getElementById('list'));
dragSort.onSort((order) => {
    console.log('New order:', order);
});
```

**关键点：**
1. 使用 HTML5 Drag and Drop API
2. `dragover` 中判断插入位置
3. `dragend` 中触发回调通知排序变化
