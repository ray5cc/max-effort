### 1. 实现简单的 URL 短链服务

**题目：** 实现一个 URL 缩短服务，支持生成短链和还原原始链接。

**思路：**
- 使用 base62 编码生成短链
- 哈希表存储映射关系
- 考虑冲突处理

**实现（Python）：**

```python
import hashlib
import random
import string

class URLShortener:
    def __init__(self):
        self.url_to_code = {}  # 长链 -> 短码
        self.code_to_url = {}  # 短码 -> 长链
        self.base_url = "http://short.url/"
        self.counter = 0

    def _generate_code(self, url):
        """生成短码"""
        # 方法1: 使用计数器 + base62
        return self._base62_encode(self.counter)

    def _base62_encode(self, num):
        """Base62 编码"""
        chars = string.digits + string.ascii_letters  # 0-9 a-z A-Z
        if num == 0:
            return chars[0]

        result = []
        while num:
            result.append(chars[num % 62])
            num //= 62

        return ''.join(reversed(result))

    def _base62_decode(self, code):
        """Base62 解码"""
        chars = string.digits + string.ascii_letters
        num = 0
        for char in code:
            num = num * 62 + chars.index(char)
        return num

    def shorten(self, long_url):
        """缩短 URL"""
        # 如果已存在，直接返回
        if long_url in self.url_to_code:
            return self.base_url + self.url_to_code[long_url]

        # 生成新的短码
        code = self._generate_code(long_url)
        self.counter += 1

        # 存储映射
        self.url_to_code[long_url] = code
        self.code_to_url[code] = long_url

        return self.base_url + code

    def expand(self, short_url):
        """还原 URL"""
        code = short_url.replace(self.base_url, "")
        return self.code_to_url.get(code, None)

# 测试

# 多个 URL

print(f"Short URL: {short2}")

# Hash-based 版本
class URLShortenerHash:
    def __init__(self):
        self.code_to_url = {}
        self.base_url = "http://short.url/"

    def shorten(self, long_url):
        # 使用 MD5 的前 6 位
        hash_value = hashlib.md5(long_url.encode()).hexdigest()[:6]

        # 处理冲突
        code = hash_value
        counter = 0
        while code in self.code_to_url and self.code_to_url[code] != long_url:
            code = hash_value + str(counter)
            counter += 1

        self.code_to_url[code] = long_url
        return self.base_url + code

    def expand(self, short_url):
        code = short_url.replace(self.base_url, "")
        return self.code_to_url.get(code)

# 测试

print(f"Expanded: {shortener_hash.expand(short)}")
```

2. 哈希表存储双向映射

### 2. 实现一致性哈希

**题目：** 实现一致性哈希算法，用于分布式缓存节点选择。

**思路：**
- 使用哈希环
- 虚拟节点解决数据倾斜

**实现：**

```python
import hashlib
import bisect

class ConsistentHash:
    def __init__(self, nodes=None, virtual_nodes=150):
        """
        nodes: 物理节点列表
        virtual_nodes: 每个物理节点对应的虚拟节点数
        """
        self.virtual_nodes = virtual_nodes
        self.ring = {}  # 哈希环: hash -> 节点名
        self.sorted_keys = []  # 有序的哈希值列表
        self.nodes = set()

        if nodes:
            for node in nodes:
                self.add_node(node)

    def _hash(self, key):
        """计算哈希值"""
        return int(hashlib.md5(str(key).encode()).hexdigest(), 16)

    def add_node(self, node):
        """添加节点"""
        self.nodes.add(node)

        for i in range(self.virtual_nodes):
            virtual_key = f"{node}#{i}"
            hash_value = self._hash(virtual_key)

            self.ring[hash_value] = node
            bisect.insort(self.sorted_keys, hash_value)

    def remove_node(self, node):
        """移除节点"""
        self.nodes.discard(node)

        for i in range(self.virtual_nodes):
            virtual_key = f"{node}#{i}"
            hash_value = self._hash(virtual_key)

            del self.ring[hash_value]
            self.sorted_keys.remove(hash_value)

    def get_node(self, key):
        """获取键对应的节点"""
        if not self.ring:
            return None

        hash_value = self._hash(key)

        # 查找第一个大于等于 hash_value 的节点
        idx = bisect.bisect(self.sorted_keys, hash_value)

        # 如果超出范围，环绕到第一个节点
        if idx == len(self.sorted_keys):
            idx = 0

        return self.ring[self.sorted_keys[idx]]

# 测试

# 添加节点
ch.add_node("server1")
ch.add_node("server2")
ch.add_node("server3")

# 测试数据分布
keys = [f"key{i}" for i in range(100)]
distribution = {}

for key in keys:
    node = ch.get_node(key)
    distribution[node] = distribution.get(node, 0) + 1

print("Initial distribution:")
for node, count in sorted(distribution.items()):
    print(f"{node}: {count} keys")

# 添加新节点
ch.add_node("server4")
print("\nAfter adding server4:")

new_distribution = {}
for key in keys:
    node = ch.get_node(key)
    new_distribution[node] = new_distribution.get(node, 0) + 1

for node, count in sorted(new_distribution.items()):
    print(f"{node}: {count} keys")

# 移除节点
ch.remove_node("server2")
print("\nAfter removing server2:")

final_distribution = {}
for key in keys:
    node = ch.get_node(key)
    final_distribution[node] = final_distribution.get(node, 0) + 1

for node, count in sorted(final_distribution.items()):
    print(f"{node}: {count} keys")
```

**关键点：**
1. 使用虚拟节点解决负载不均
2. 有序列表 + 二分查找
3. 添加/删除节点时只影响部分数据

### 3. 实现简单的负载均衡器

**题目：** 实现支持多种策略的负载均衡器。

**实现：**

```python
import random
from abc import ABC, abstractmethod

class LoadBalancer(ABC):
    def __init__(self, servers):
        self.servers = servers
        self.current_index = 0

    @abstractmethod
    def select_server(self):
        """选择服务器"""
        pass

# 1. 轮询（Round Robin）
class RoundRobinLB(LoadBalancer):
    def select_server(self):
        if not self.servers:
            return None

        server = self.servers[self.current_index]
        self.current_index = (self.current_index + 1) % len(self.servers)
        return server

# 2. 随机（Random）
class RandomLB(LoadBalancer):
    def select_server(self):
        if not self.servers:
            return None
        return random.choice(self.servers)

# 3. 加权轮询（Weighted Round Robin）
class WeightedRoundRobinLB:
    def __init__(self, servers_with_weights):
        """
        servers_with_weights: [(server, weight), ...]
        """
        self.servers = []
        self.weights = []
        self.current_weight = 0
        self.max_weight = 0
        self.gcd_weight = 0

        for server, weight in servers_with_weights:
            self.servers.append(server)
            self.weights.append(weight)
            self.max_weight = max(self.max_weight, weight)

        self.gcd_weight = self._gcd_list(self.weights)
        self.current_index = -1

    def _gcd(self, a, b):
        while b:
            a, b = b, a % b
        return a

    def _gcd_list(self, weights):
        result = weights[0]
        for w in weights[1:]:
            result = self._gcd(result, w)
        return result

    def select_server(self):
        while True:
            self.current_index = (self.current_index + 1) % len(self.servers)

            if self.current_index == 0:
                self.current_weight = self.current_weight - self.gcd_weight
                if self.current_weight <= 0:
                    self.current_weight = self.max_weight

            if self.weights[self.current_index] >= self.current_weight:
                return self.servers[self.current_index]

# 4. 最少连接（Least Connections）
class LeastConnectionsLB:
    def __init__(self, servers):
        self.servers = servers
        self.connections = {server: 0 for server in servers}

    def select_server(self):
        if not self.servers:
            return None

        # 选择连接数最少的服务器
        return min(self.connections, key=self.connections.get)

    def connect(self, server):
        """增加连接数"""
        self.connections[server] += 1

    def disconnect(self, server):
        """减少连接数"""
        if self.connections[server] > 0:
            self.connections[server] -= 1

# 5. IP 哈希（IP Hash）
class IPHashLB(LoadBalancer):
    def select_server(self, client_ip):
        if not self.servers:
            return None

        hash_value = hash(client_ip)
        index = hash_value % len(self.servers)
        return self.servers[index]

# 测试
print("=== Round Robin ===")
rr_lb = RoundRobinLB(["server1", "server2", "server3"])
for _ in range(5):
    print(rr_lb.select_server())

print("\n=== Random ===")
random_lb = RandomLB(["server1", "server2", "server3"])
for _ in range(5):
    print(random_lb.select_server())

print("\n=== Weighted Round Robin ===")
wrr_lb = WeightedRoundRobinLB([
    ("server1", 5),
    ("server2", 1),
    ("server3", 1)
])
for _ in range(10):
    print(wrr_lb.select_server())

print("\n=== Least Connections ===")
lc_lb = LeastConnectionsLB(["server1", "server2", "server3"])
for _ in range(5):
    server = lc_lb.select_server()
    print(f"Selected: {server}")
    lc_lb.connect(server)

print("\n=== IP Hash ===")
ip_lb = IPHashLB(["server1", "server2", "server3"])
for ip in ["192.168.1.1", "192.168.1.2", "192.168.1.1"]:
    print(f"{ip} -> {ip_lb.select_server(ip)}")
```

**关键点：**
1. 轮询：简单，分布均匀
2. 加权轮询：考虑服务器性能差异
3. 最少连接：动态负载均衡
4. IP 哈希：会话保持

### 4. 实现限流器（多种算法）

**题目：** 实现固定窗口、滑动窗口、令牌桶、漏桶等限流算法。

**实现：**

```python
import time
from collections import deque

# 1. 固定窗口
class FixedWindowRateLimiter:
    def __init__(self, capacity, window_size):
        """
        capacity: 窗口内最大请求数
        window_size: 窗口大小（秒）
        """
        self.capacity = capacity
        self.window_size = window_size
        self.count = 0
        self.window_start = time.time()

    def allow_request(self):
        now = time.time()

        # 新窗口
        if now - self.window_start >= self.window_size:
            self.count = 0
            self.window_start = now

        if self.count < self.capacity:
            self.count += 1
            return True

        return False

# 2. 滑动窗口
class SlidingWindowRateLimiter:
    def __init__(self, capacity, window_size):
        self.capacity = capacity
        self.window_size = window_size
        self.requests = deque()

    def allow_request(self):
        now = time.time()

        # 移除过期请求
        while self.requests and now - self.requests[0] > self.window_size:
            self.requests.popleft()

        if len(self.requests) < self.capacity:
            self.requests.append(now)
            return True

        return False

# 3. 令牌桶
class TokenBucketRateLimiter:
    def __init__(self, capacity, refill_rate):
        """
        capacity: 桶容量
        refill_rate: 每秒补充令牌数
        """
        self.capacity = capacity
        self.refill_rate = refill_rate
        self.tokens = capacity
        self.last_refill = time.time()

    def _refill(self):
        now = time.time()
        elapsed = now - self.last_refill
        tokens_to_add = elapsed * self.refill_rate

        self.tokens = min(self.capacity, self.tokens + tokens_to_add)
        self.last_refill = now

    def allow_request(self, tokens=1):
        self._refill()

        if self.tokens >= tokens:
            self.tokens -= tokens
            return True

        return False

# 4. 漏桶
class LeakyBucketRateLimiter:
    def __init__(self, capacity, leak_rate):
        """
        capacity: 桶容量
        leak_rate: 每秒漏出请求数
        """
        self.capacity = capacity
        self.leak_rate = leak_rate
        self.queue = deque()
        self.last_leak = time.time()

    def _leak(self):
        now = time.time()
        elapsed = now - self.last_leak
        leaks = int(elapsed * self.leak_rate)

        for _ in range(min(leaks, len(self.queue))):
            self.queue.popleft()

        self.last_leak = now

    def allow_request(self):
        self._leak()

        if len(self.queue) < self.capacity:
            self.queue.append(time.time())
            return True

        return False

# 测试
def test_rate_limiter(limiter, name, requests_per_second):
    print(f"\n=== {name} ===")
    allowed = 0
    denied = 0

    for i in range(requests_per_second * 2):
        if limiter.allow_request():
            allowed += 1
        else:
            denied += 1
        time.sleep(1 / requests_per_second)

    print(f"Allowed: {allowed}, Denied: {denied}")

# 测试各种限流器
test_rate_limiter(
    FixedWindowRateLimiter(10, 1),
    "Fixed Window (10 req/s)",
    20
)

test_rate_limiter(
    SlidingWindowRateLimiter(10, 1),
    "Sliding Window (10 req/s)",
    20
)

test_rate_limiter(
    TokenBucketRateLimiter(10, 10),
    "Token Bucket (10 req/s)",
    20
)

test_rate_limiter(
    LeakyBucketRateLimiter(10, 10),
    "Leaky Bucket (10 req/s)",
    20
)
```

**关键点：**
1. 固定窗口：简单但有临界问题
2. 滑动窗口：精确但空间开销大
3. 令牌桶：允许突发流量
4. 漏桶：平滑流量

### 5. 实现分布式唯一 ID 生成器

**题目：** 实现 Snowflake 算法的分布式 ID 生成器。

**实现：**

```python
import time
import threading

class SnowflakeIDGenerator:
    def __init__(self, datacenter_id, worker_id):
        """
        Snowflake ID 结构（64位）:
        - 1位: 符号位（0）
        - 41位: 时间戳（毫秒）
        - 5位: 数据中心ID
        - 5位: 机器ID
        - 12位: 序列号
        """
        self.datacenter_id = datacenter_id
        self.worker_id = worker_id
        self.sequence = 0
        self.last_timestamp = -1
        self.lock = threading.Lock()

        # 位数配置
        self.datacenter_id_bits = 5
        self.worker_id_bits = 5
        self.sequence_bits = 12

        # 最大值
        self.max_datacenter_id = -1 ^ (-1 << self.datacenter_id_bits)
        self.max_worker_id = -1 ^ (-1 << self.worker_id_bits)
        self.max_sequence = -1 ^ (-1 << self.sequence_bits)

        # 位移量
        self.worker_id_shift = self.sequence_bits
        self.datacenter_id_shift = self.sequence_bits + self.worker_id_bits
        self.timestamp_shift = (self.sequence_bits + self.worker_id_bits +
                               self.datacenter_id_bits)

        # 起始时间戳（2020-01-01）
        self.epoch = 1577836800000

        # 验证参数
        if datacenter_id > self.max_datacenter_id or datacenter_id < 0:
            raise ValueError(f"Datacenter ID must be between 0 and {self.max_datacenter_id}")
        if worker_id > self.max_worker_id or worker_id < 0:
            raise ValueError(f"Worker ID must be between 0 and {self.max_worker_id}")

    def _current_millis(self):
        return int(time.time() * 1000)

    def _wait_next_millis(self, last_timestamp):
        timestamp = self._current_millis()
        while timestamp <= last_timestamp:
            timestamp = self._current_millis()
        return timestamp

    def next_id(self):
        with self.lock:
            timestamp = self._current_millis()

            # 时钟回拨检测
            if timestamp < self.last_timestamp:
                raise Exception(f"Clock moved backwards. Refusing to generate id for "
                              f"{self.last_timestamp - timestamp} milliseconds")

            if timestamp == self.last_timestamp:
                # 同一毫秒内，序列号递增
                self.sequence = (self.sequence + 1) & self.max_sequence
                if self.sequence == 0:
                    # 序列号溢出，等待下一毫秒
                    timestamp = self._wait_next_millis(self.last_timestamp)
            else:
                # 新的毫秒，序列号重置
                self.sequence = 0

            self.last_timestamp = timestamp

            # 组装 ID
            return (
                ((timestamp - self.epoch) << self.timestamp_shift) |
                (self.datacenter_id << self.datacenter_id_shift) |
                (self.worker_id << self.worker_id_shift) |
                self.sequence
            )

    def parse_id(self, id_value):
        """解析 ID"""
        timestamp = ((id_value >> self.timestamp_shift) + self.epoch)
        datacenter_id = (id_value >> self.datacenter_id_shift) & self.max_datacenter_id
        worker_id = (id_value >> self.worker_id_shift) & self.max_worker_id
        sequence = id_value & self.max_sequence

        return {
            'timestamp': timestamp,
            'datacenter_id': datacenter_id,
            'worker_id': worker_id,
            'sequence': sequence
        }

# 测试
generator = SnowflakeIDGenerator(datacenter_id=1, worker_id=1)

# 生成 ID
ids = [generator.next_id() for _ in range(10)]
print("Generated IDs:")
for id_value in ids:
    print(f"ID: {id_value}")

# 解析 ID
print(f"\nParsing ID: {ids[0]}")
parsed = generator.parse_id(ids[0])
for key, value in parsed.items():
    print(f"{key}: {value}")

# 并发测试
import concurrent.futures

def generate_ids(count):
    return [generator.next_id() for _ in range(count)]

with concurrent.futures.ThreadPoolExecutor(max_workers=10) as executor:
    futures = [executor.submit(generate_ids, 100) for _ in range(10)]
    results = [future.result() for future in futures]
    all_ids = [id_val for result in results for id_val in result]

    # 检查唯一性
    print(f"\nTotal IDs: {len(all_ids)}")
    print(f"Unique IDs: {len(set(all_ids))}")
    print(f"All unique: {len(all_ids) == len(set(all_ids))}")
```

**关键点：**
1. 时间戳保证趋势递增
2. 机器ID保证分布式唯一性
3. 序列号支持同一毫秒多个ID
4. 处理时钟回拨问题

### 6. 实现分布式锁

**题目：** 基于 Redis 实现分布式锁，支持重入、超时、续期。

**实现：**

```python
import time
import uuid
import threading

class RedisDistributedLock:
    """模拟 Redis 分布式锁（使用字典模拟 Redis）"""

    # 模拟 Redis 存储
    _redis_storage = {}
    _lock = threading.Lock()

    def __init__(self, lock_key, expire_time=10):
        self.lock_key = lock_key
        self.lock_value = str(uuid.uuid4())
        self.expire_time = expire_time
        self.acquired = False
        self.renew_thread = None
        self.stop_renew = False

    def acquire(self, blocking=True, timeout=None):
        """获取锁"""
        start_time = time.time()

        while True:
            # 尝试获取锁
            if self._try_acquire():
                self.acquired = True
                # 启动自动续期
                self._start_auto_renew()
                return True

            if not blocking:
                return False

            if timeout and (time.time() - start_time) >= timeout:
                return False

            time.sleep(0.01)  # 短暂休眠后重试

    def _try_acquire(self):
        """尝试获取锁（SET NX EX）"""
        with self._lock:
            current_time = time.time()

            # 检查锁是否存在且未过期
            if self.lock_key in self._redis_storage:
                lock_data = self._redis_storage[self.lock_key]
                if lock_data['expire_time'] > current_time:
                    # 检查是否是同一个客户端（可重入）
                    if lock_data['value'] == self.lock_value:
                        lock_data['count'] += 1
                        return True
                    return False
                else:
                    # 锁已过期，删除
                    del self._redis_storage[self.lock_key]

            # 设置新锁
            self._redis_storage[self.lock_key] = {
                'value': self.lock_value,
                'expire_time': current_time + self.expire_time,
                'count': 1
            }
            return True

    def release(self):
        """释放锁"""
        if not self.acquired:
            return False

        # 停止自动续期
        self._stop_auto_renew()

        with self._lock:
            if self.lock_key not in self._redis_storage:
                return False

            lock_data = self._redis_storage[self.lock_key]

            # 检查是否是自己的锁
            if lock_data['value'] != self.lock_value:
                return False

            # 可重入锁计数递减
            lock_data['count'] -= 1
            if lock_data['count'] == 0:
                del self._redis_storage[self.lock_key]

        self.acquired = False
        return True

    def renew(self):
        """续期"""
        with self._lock:
            if self.lock_key not in self._redis_storage:
                return False

            lock_data = self._redis_storage[self.lock_key]

            if lock_data['value'] != self.lock_value:
                return False

            lock_data['expire_time'] = time.time() + self.expire_time
            return True

    def _start_auto_renew(self):
        """启动自动续期线程"""
        self.stop_renew = False
        self.renew_thread = threading.Thread(target=self._auto_renew_loop)
        self.renew_thread.daemon = True
        self.renew_thread.start()

    def _auto_renew_loop(self):
        """自动续期循环"""
        while not self.stop_renew:
            time.sleep(self.expire_time / 3)  # 每 1/3 过期时间续期一次
            if not self.stop_renew:
                self.renew()

    def _stop_auto_renew(self):
        """停止自动续期"""
        self.stop_renew = True
        if self.renew_thread:
            self.renew_thread.join(timeout=1)

    def __enter__(self):
        """支持 with 语句"""
        if not self.acquire():
            raise RuntimeError("Failed to acquire lock")
        return self

    def __exit__(self, exc_type, exc_val, exc_tb):
        """自动释放锁"""
        self.release()
```

## 🎯 场景题

### 7. 实现简单的消息队列

**题目：** 实现支持发布订阅、优先级队列的消息队列。

**实现：**

```python
import heapq
import threading
import time
from collections import defaultdict, deque
from enum import Enum

class Message:
    def __init__(self, content, priority=0, delay=0):
        self.content = content
        self.priority = priority
        self.timestamp = time.time() + delay
        self.id = id(self)

    def __lt__(self, other):
        # 优先级高的先处理，相同优先级按时间排序
        if self.priority != other.priority:
            return self.priority > other.priority
        return self.timestamp < other.timestamp

class MessageQueue:
    def __init__(self):
        self.queue = []  # 优先级队列
        self.subscribers = defaultdict(list)  # topic -> [callbacks]
        self.lock = threading.Lock()
        self.not_empty = threading.Condition(self.lock)
        self.running = True

    def publish(self, topic, content, priority=0, delay=0):
        """发布消息"""
        message = Message(content, priority, delay)

        with self.lock:
            heapq.heappush(self.queue, (topic, message))
            self.not_empty.notify()

    def subscribe(self, topic, callback):
        """订阅主题"""
        with self.lock:
            self.subscribers[topic].append(callback)

    def unsubscribe(self, topic, callback):
        """取消订阅"""
        with self.lock:
            if callback in self.subscribers[topic]:
                self.subscribers[topic].remove(callback)

    def consume(self, timeout=None):
        """消费消息"""
        with self.not_empty:
            while self.running:
                if not self.queue:
                    if timeout:
                        self.not_empty.wait(timeout)
                        if not self.queue:
                            return None
                    else:
                        self.not_empty.wait()
                        continue

                # 检查延迟消息
                topic, message = self.queue[0]
                if message.timestamp > time.time():
                    wait_time = message.timestamp - time.time()
                    self.not_empty.wait(wait_time)
                    continue

                # 取出消息
                heapq.heappop(self.queue)

                # 通知订阅者
                if topic in self.subscribers:
                    for callback in self.subscribers[topic]:
                        try:
                            callback(message.content)
                        except Exception as e:
                            print(f"Error in callback: {e}")

                return topic, message.content

    def start_worker(self):
        """启动后台工作线程"""
        def worker():
            while self.running:
                self.consume(timeout=1)

        thread = threading.Thread(target=worker, daemon=True)
        thread.start()
        return thread

    def stop(self):
        """停止队列"""
        with self.lock:
            self.running = False
            self.not_empty.notify_all()

# 测试
mq = MessageQueue()

# 订阅主题
def handle_order(message):
    print(f"[Order Handler] Received: {message}")

def handle_payment(message):
    print(f"[Payment Handler] Received: {message}")

mq.subscribe("orders", handle_order)
mq.subscribe("payments", handle_payment)

# 启动工作线程
mq.start_worker()

# 发布消息
print("Publishing messages...")
mq.publish("orders", "Order #1", priority=1)
mq.publish("payments", "Payment #1", priority=2)  # 高优先级
mq.publish("orders", "Order #2", priority=1)
mq.publish("orders", "Urgent Order #3", priority=5)  # 最高优先级
mq.publish("payments", "Delayed Payment", delay=2)  # 延迟2秒

# 等待处理

mq.stop()
```

**关键点：**
1. 优先级队列使用堆实现
2. 延迟消息支持
3. 发布订阅模式
4. 线程安全
