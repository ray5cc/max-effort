### 1. 查询第 N 高的薪水

**题目：** 编写 SQL 查询，获取 Employee 表中第 N 高的薪水。

```sql
-- 表结构
CREATE TABLE Employee (
    id INT PRIMARY KEY,
    salary INT
);

-- 示例数据
INSERT INTO Employee VALUES
(1, 100), (2, 200), (3, 300), (4, 300), (5, 400);
```

**思路：**
- 使用 OFFSET 和 LIMIT
- 处理重复薪水（DISTINCT）
- 考虑不存在的情况

**实现（MySQL）：**

```sql
-- 方法1: 使用 LIMIT OFFSET
SELECT DISTINCT salary
FROM Employee
ORDER BY salary DESC
LIMIT 1 OFFSET N-1;

-- 方法2: 使用子查询
SELECT DISTINCT salary
FROM Employee e1
WHERE (SELECT COUNT(DISTINCT salary)
       FROM Employee e2
       WHERE e2.salary > e1.salary) = N-1;

-- 方法3: 使用窗口函数（推荐）
WITH RankedSalaries AS (
    SELECT salary,
           DENSE_RANK() OVER (ORDER BY salary DESC) as rank_num
    FROM Employee
)
SELECT DISTINCT salary
FROM RankedSalaries
WHERE rank_num = N;

-- 创建函数
DELIMITER $$
CREATE FUNCTION getNthHighestSalary(N INT) RETURNS INT
BEGIN
    DECLARE result INT;

    SELECT DISTINCT salary INTO result
    FROM (
        SELECT salary,
               DENSE_RANK() OVER (ORDER BY salary DESC) as rank_num
        FROM Employee
    ) ranked
    WHERE rank_num = N;

    RETURN result;
END$$
DELIMITER ;

-- 测试
SELECT getNthHighestSalary(2); -- 返回 300
SELECT getNthHighestSalary(3); -- 返回 200
```

**关键点：**
1. 使用 `DISTINCT` 处理重复值
2. `DENSE_RANK()` 连续排名（300, 300 都是第2名）
3. `RANK()` 跳跃排名（300, 300 是第2名，下一个是第4名）

### 2. 删除重复的电子邮箱

**题目：** 删除 Person 表中重复的邮箱，保留 ID 最小的记录。

```sql
-- 表结构
CREATE TABLE Person (
    id INT PRIMARY KEY,
    email VARCHAR(255)
);

-- 示例数据
INSERT INTO Person VALUES
(1, 'john@example.com'),
(2, 'bob@example.com'),
(3, 'john@example.com');
```

**思路：**
- 找到每个邮箱的最小 ID
- 删除 ID 不是最小的记录

**实现：**

```sql
-- 方法1: 使用 DELETE JOIN
DELETE p1 FROM Person p1
INNER JOIN Person p2
ON p1.email = p2.email AND p1.id > p2.id;

-- 方法2: 使用子查询
DELETE FROM Person
WHERE id NOT IN (
    SELECT * FROM (
        SELECT MIN(id)
        FROM Person
        GROUP BY email
    ) AS min_ids
);

-- 方法3: 使用窗口函数（MySQL 8.0+）
DELETE FROM Person
WHERE id IN (
    SELECT id FROM (
        SELECT id,
               ROW_NUMBER() OVER (PARTITION BY email ORDER BY id) as rn
        FROM Person
    ) ranked
    WHERE rn > 1
);

-- 验证
SELECT * FROM Person;
-- 结果: (1, 'john@example.com'), (2, 'bob@example.com')
```

**关键点：**
1. 自连接删除效率较高
2. 注意子查询需要别名
3. `ROW_NUMBER()` 为每个分区编号

### 3. 查询连续出现的数字

**题目：** 找出 Logs 表中连续出现至少 3 次的数字。

```sql
-- 表结构
CREATE TABLE Logs (
    id INT PRIMARY KEY AUTO_INCREMENT,
    num INT
);

-- 示例数据
INSERT INTO Logs (num) VALUES
(1), (1), (1), (2), (1), (1), (2), (2);
```

**思路：**
- 使用自连接检查连续性
- 或使用窗口函数

**实现：**

```sql
-- 方法1: 三表自连接
SELECT DISTINCT l1.num AS ConsecutiveNums
FROM Logs l1
JOIN Logs l2 ON l1.id = l2.id - 1 AND l1.num = l2.num
JOIN Logs l3 ON l2.id = l3.id - 1 AND l2.num = l3.num;

-- 方法2: 使用窗口函数（推荐）
WITH Numbered AS (
    SELECT num,
           LAG(num, 1) OVER (ORDER BY id) as prev_num,
           LAG(num, 2) OVER (ORDER BY id) as prev_prev_num
    FROM Logs
)
SELECT DISTINCT num AS ConsecutiveNums
FROM Numbered
WHERE num = prev_num AND num = prev_prev_num;

-- 方法3: 使用变量（MySQL 5.7）
SELECT DISTINCT num AS ConsecutiveNums
FROM (
    SELECT num,
           @count := IF(@prev = num, @count + 1, 1) AS count,
           @prev := num
    FROM Logs, (SELECT @count := 0, @prev := NULL) init
    ORDER BY id
) counted
WHERE count >= 3;

-- 测试
SELECT * FROM (
    SELECT num, COUNT(*) as consecutive_count
    FROM (
        SELECT num,
               id - ROW_NUMBER() OVER (PARTITION BY num ORDER BY id) as grp
        FROM Logs
    ) grouped
    GROUP BY num, grp
    HAVING COUNT(*) >= 3
) result;
```

**关键点：**
1. `LAG()` 获取前一行的值
2. `LEAD()` 获取后一行的值
3. 使用分组技巧：`id - ROW_NUMBER()` 相同表示连续

### 4. 部门工资前三名

**题目：** 查询每个部门工资前三名的员工。

```sql
-- 表结构
CREATE TABLE Employee (
    id INT PRIMARY KEY,
    name VARCHAR(50),
    salary INT,
    department_id INT
);

CREATE TABLE Department (
    id INT PRIMARY KEY,
    name VARCHAR(50)
);

-- 示例数据
INSERT INTO Department VALUES (1, 'IT'), (2, 'Sales');
INSERT INTO Employee VALUES
(1, 'Joe', 70000, 1),
(2, 'Henry', 80000, 2),
(3, 'Sam', 60000, 2),
(4, 'Max', 90000, 1),
(5, 'Janet', 69000, 1),
(6, 'Randy', 85000, 1),
(7, 'Will', 70000, 1);
```

**实现：**

```sql
-- 方法1: 使用窗口函数（推荐）
SELECT d.name AS Department,
       e.name AS Employee,
       e.salary AS Salary
FROM (
    SELECT *,
           DENSE_RANK() OVER (PARTITION BY department_id
                              ORDER BY salary DESC) as rank_num
    FROM Employee
) e
JOIN Department d ON e.department_id = d.id
WHERE e.rank_num <= 3
ORDER BY d.name, e.salary DESC;

-- 方法2: 使用子查询
SELECT d.name AS Department,
       e1.name AS Employee,
       e1.salary AS Salary
FROM Employee e1
JOIN Department d ON e1.department_id = d.id
WHERE (
    SELECT COUNT(DISTINCT e2.salary)
    FROM Employee e2
    WHERE e2.department_id = e1.department_id
      AND e2.salary > e1.salary
) < 3
ORDER BY d.name, e1.salary DESC;

-- 方法3: 使用 CTE
WITH RankedEmployees AS (
    SELECT e.*,
           d.name as dept_name,
           DENSE_RANK() OVER (PARTITION BY e.department_id
                              ORDER BY e.salary DESC) as salary_rank
    FROM Employee e
    JOIN Department d ON e.department_id = d.id
)
SELECT dept_name AS Department,
       name AS Employee,
       salary AS Salary
FROM RankedEmployees
WHERE salary_rank <= 3
ORDER BY dept_name, salary DESC;
```

**关键点：**
1. `PARTITION BY` 按部门分组排名
2. `DENSE_RANK()` 处理并列情况
3. CTE 提高可读性

### 5. 查询每个用户的累计登录天数

**题目：** 统计每个用户的累计登录天数（去重）。

```sql
-- 表结构
CREATE TABLE LoginLog (
    user_id INT,
    login_date DATE
);

-- 示例数据
INSERT INTO LoginLog VALUES
(1, '2024-01-01'),
(1, '2024-01-01'),
(1, '2024-01-02'),
(2, '2024-01-01'),
(1, '2024-01-03'),
(2, '2024-01-02');
```

**实现：**

```sql
-- 方法1: 简单统计
SELECT user_id,
       COUNT(DISTINCT login_date) as total_login_days
FROM LoginLog
GROUP BY user_id;

-- 方法2: 带累计计算
SELECT user_id,
       login_date,
       ROW_NUMBER() OVER (PARTITION BY user_id
                          ORDER BY login_date) as login_day_number
FROM (
    SELECT DISTINCT user_id, login_date
    FROM LoginLog
) unique_logins
ORDER BY user_id, login_date;

-- 方法3: 查询连续登录天数
WITH DailyLogins AS (
    SELECT DISTINCT user_id, login_date
    FROM LoginLog
),
Grouped AS (
    SELECT user_id,
           login_date,
           DATE_SUB(login_date, INTERVAL
                    ROW_NUMBER() OVER (PARTITION BY user_id
                                       ORDER BY login_date) DAY) as grp
    FROM DailyLogins
)
SELECT user_id,
       MIN(login_date) as streak_start,
       MAX(login_date) as streak_end,
       COUNT(*) as consecutive_days
FROM Grouped
GROUP BY user_id, grp
HAVING COUNT(*) >= 3  -- 连续3天以上
ORDER BY user_id, streak_start;

-- 测试数据
INSERT INTO LoginLog VALUES
(1, '2024-01-04'),
(1, '2024-01-05'),
(1, '2024-01-06');

-- 结果会显示用户1有连续登录记录
```

**关键点：**
1. `DISTINCT` 去除同一天多次登录
2. 连续登录检测：日期 - 行号 = 常量
3. `DATE_SUB` 进行日期计算

### 6. 查询互相关注的用户对

**题目：** 找出互相关注的用户对（A 关注 B，B 也关注 A）。

```sql
-- 表结构
CREATE TABLE Follow (
    follower_id INT,
    followee_id INT,
    PRIMARY KEY (follower_id, followee_id)
);

-- 示例数据
INSERT INTO Follow VALUES
(1, 2), (2, 1),  -- 互相关注
(1, 3), (3, 1),  -- 互相关注
(2, 3),          -- 单向关注
(4, 1);          -- 单向关注
```

**实现：**

```sql
-- 方法1: 自连接
SELECT DISTINCT
    LEAST(f1.follower_id, f1.followee_id) as user1,
    GREATEST(f1.follower_id, f1.followee_id) as user2
FROM Follow f1
JOIN Follow f2
ON f1.follower_id = f2.followee_id
AND f1.followee_id = f2.follower_id
WHERE f1.follower_id < f1.followee_id;  -- 避免重复

-- 方法2: 使用 EXISTS
SELECT DISTINCT
    LEAST(follower_id, followee_id) as user1,
    GREATEST(follower_id, followee_id) as user2
FROM Follow f1
WHERE EXISTS (
    SELECT 1 FROM Follow f2
    WHERE f2.follower_id = f1.followee_id
      AND f2.followee_id = f1.follower_id
)
AND follower_id < followee_id;

-- 扩展: 查询共同关注
SELECT f1.follower_id as user_id,
       COUNT(DISTINCT f1.followee_id) as common_followees
FROM Follow f1
JOIN Follow f2
ON f1.followee_id = f2.followee_id
WHERE f1.follower_id = 1  -- 用户1
  AND f2.follower_id = 2  -- 用户2
  AND f1.follower_id != f2.follower_id;

-- 扩展: 推荐关注（朋友的朋友）
SELECT DISTINCT f2.followee_id as recommended_user
FROM Follow f1
JOIN Follow f2 ON f1.followee_id = f2.follower_id
WHERE f1.follower_id = 1  -- 当前用户
  AND f2.followee_id != 1  -- 不推荐自己
  AND f2.followee_id NOT IN (  -- 尚未关注
      SELECT followee_id
      FROM Follow
      WHERE follower_id = 1
  );
```

**关键点：**
1. `LEAST/GREATEST` 避免重复对
2. 使用 `EXISTS` 提高性能
3. 社交网络常见查询模式

### 7. 计算移动平均值

**题目：** 计算每天的 7 日移动平均销售额。

```sql
-- 表结构
CREATE TABLE Sales (
    sale_date DATE,
    amount DECIMAL(10, 2)
);

-- 示例数据
INSERT INTO Sales VALUES
('2024-01-01', 100),
('2024-01-02', 150),
('2024-01-03', 200),
('2024-01-04', 180),
('2024-01-05', 220),
('2024-01-06', 190),
('2024-01-07', 210),
('2024-01-08', 230);
```

**实现：**

```sql
-- 方法1: 使用窗口函数（推荐）
SELECT sale_date,
       amount,
       AVG(amount) OVER (
           ORDER BY sale_date
           ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
       ) as moving_avg_7d,
       SUM(amount) OVER (
           ORDER BY sale_date
           ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
       ) as moving_sum_7d
FROM Sales
ORDER BY sale_date;

-- 方法2: 自连接
SELECT s1.sale_date,
       s1.amount,
       AVG(s2.amount) as moving_avg_7d
FROM Sales s1
JOIN Sales s2
ON s2.sale_date BETWEEN DATE_SUB(s1.sale_date, INTERVAL 6 DAY)
                    AND s1.sale_date
GROUP BY s1.sale_date, s1.amount
ORDER BY s1.sale_date;

-- 扩展: 指数移动平均（EMA）
WITH RECURSIVE EMA AS (
    SELECT sale_date,
           amount,
           amount as ema,
           1 as rn
    FROM Sales
    WHERE sale_date = (SELECT MIN(sale_date) FROM Sales)

    UNION ALL

    SELECT s.sale_date,
           s.amount,
           0.2 * s.amount + 0.8 * e.ema as ema,  -- alpha = 0.2
           e.rn + 1
    FROM Sales s
    JOIN EMA e ON s.sale_date > e.sale_date
    WHERE s.sale_date = (
        SELECT MIN(sale_date)
        FROM Sales
        WHERE sale_date > e.sale_date
    )
)
SELECT sale_date,
       amount,
       ROUND(ema, 2) as exponential_moving_avg
FROM EMA
ORDER BY sale_date;

-- 扩展: 同比、环比增长率
SELECT sale_date,
       amount,
       LAG(amount, 1) OVER (ORDER BY sale_date) as prev_day,
       LAG(amount, 7) OVER (ORDER BY sale_date) as prev_week,
       ROUND((amount - LAG(amount, 1) OVER (ORDER BY sale_date)) /
             LAG(amount, 1) OVER (ORDER BY sale_date) * 100, 2)
             as day_over_day_pct,
       ROUND((amount - LAG(amount, 7) OVER (ORDER BY sale_date)) /
             LAG(amount, 7) OVER (ORDER BY sale_date) * 100, 2)
             as week_over_week_pct
FROM Sales
ORDER BY sale_date;
```

**关键点：**
1. `ROWS BETWEEN` 定义窗口范围
2. `PRECEDING` 表示之前的行
3. 指数移动平均需要递归 CTE

### 8. 查询每个类别的销量占比

**题目：** 计算每个商品类别的销量和占比。

```sql
-- 表结构
CREATE TABLE Products (
    product_id INT PRIMARY KEY,
    product_name VARCHAR(100),
    category VARCHAR(50),
    price DECIMAL(10, 2)
);

CREATE TABLE Orders (
    order_id INT PRIMARY KEY,
    product_id INT,
    quantity INT,
    order_date DATE
);

-- 示例数据
INSERT INTO Products VALUES
(1, 'Laptop', 'Electronics', 1000),
(2, 'Mouse', 'Electronics', 20),
(3, 'Desk', 'Furniture', 300),
(4, 'Chair', 'Furniture', 150);

INSERT INTO Orders VALUES
(1, 1, 2, '2024-01-01'),
(2, 2, 10, '2024-01-02'),
(3, 3, 5, '2024-01-03'),
(4, 1, 1, '2024-01-04');
```

**实现：**

```sql
-- 基础查询
SELECT p.category,
       SUM(o.quantity) as total_quantity,
       SUM(o.quantity * p.price) as total_revenue
FROM Orders o
JOIN Products p ON o.product_id = p.product_id
GROUP BY p.category;

-- 计算占比
WITH CategoryStats AS (
    SELECT p.category,
           SUM(o.quantity) as category_quantity,
           SUM(o.quantity * p.price) as category_revenue
    FROM Orders o
    JOIN Products p ON o.product_id = p.product_id
    GROUP BY p.category
),
TotalStats AS (
    SELECT SUM(category_quantity) as total_quantity,
           SUM(category_revenue) as total_revenue
    FROM CategoryStats
)
SELECT cs.category,
       cs.category_quantity,
       cs.category_revenue,
       ROUND(cs.category_quantity * 100.0 / ts.total_quantity, 2)
           as quantity_percentage,
       ROUND(cs.category_revenue * 100.0 / ts.total_revenue, 2)
           as revenue_percentage
FROM CategoryStats cs
CROSS JOIN TotalStats ts
ORDER BY cs.category_revenue DESC;

-- 使用窗口函数（更简洁）
SELECT category,
       category_quantity,
       category_revenue,
       ROUND(category_quantity * 100.0 /
             SUM(category_quantity) OVER (), 2) as quantity_pct,
       ROUND(category_revenue * 100.0 /
             SUM(category_revenue) OVER (), 2) as revenue_pct,
       ROUND(SUM(category_revenue) OVER (ORDER BY category_revenue DESC) * 100.0 /
             SUM(category_revenue) OVER (), 2) as cumulative_pct
FROM (
    SELECT p.category,
           SUM(o.quantity) as category_quantity,
           SUM(o.quantity * p.price) as category_revenue
    FROM Orders o
    JOIN Products p ON o.product_id = p.product_id
    GROUP BY p.category
) stats
ORDER BY category_revenue DESC;
```

**关键点:**
1. CTE 分解复杂查询
2. `SUM() OVER ()` 计算全局总和
3. 累计占比使用 `ORDER BY` 在窗口函数中

### 9. 查询留存率

**题目：** 计算用户留存率（次日、7日、30日留存）。

```sql
-- 表结构
CREATE TABLE UserActivity (
    user_id INT,
    activity_date DATE,
    PRIMARY KEY (user_id, activity_date)
);

-- 示例数据
INSERT INTO UserActivity VALUES
(1, '2024-01-01'),
(1, '2024-01-02'),
(1, '2024-01-08'),
(2, '2024-01-01'),
(2, '2024-01-03'),
(3, '2024-01-01');
```

**实现：**

```sql
-- 计算留存率
WITH FirstActivity AS (
    SELECT user_id,
           MIN(activity_date) as first_date
    FROM UserActivity
    GROUP BY user_id
),
Retention AS (
    SELECT fa.first_date,
           COUNT(DISTINCT fa.user_id) as new_users,
           COUNT(DISTINCT CASE
               WHEN ua.activity_date = DATE_ADD(fa.first_date, INTERVAL 1 DAY)
               THEN ua.user_id
           END) as day1_retention,
           COUNT(DISTINCT CASE
               WHEN ua.activity_date = DATE_ADD(fa.first_date, INTERVAL 7 DAY)
               THEN ua.user_id
           END) as day7_retention,
           COUNT(DISTINCT CASE
               WHEN ua.activity_date = DATE_ADD(fa.first_date, INTERVAL 30 DAY)
               THEN ua.user_id
           END) as day30_retention
    FROM FirstActivity fa
    LEFT JOIN UserActivity ua
        ON fa.user_id = ua.user_id
        AND ua.activity_date IN (
            DATE_ADD(fa.first_date, INTERVAL 1 DAY),
            DATE_ADD(fa.first_date, INTERVAL 7 DAY),
            DATE_ADD(fa.first_date, INTERVAL 30 DAY)
        )
    GROUP BY fa.first_date
)
SELECT first_date,
       new_users,
       day1_retention,
       ROUND(day1_retention * 100.0 / new_users, 2) as day1_retention_pct,
       day7_retention,
       ROUND(day7_retention * 100.0 / new_users, 2) as day7_retention_pct,
       day30_retention,
       ROUND(day30_retention * 100.0 / new_users, 2) as day30_retention_pct
FROM Retention
ORDER BY first_date;

-- 通用留存查询（任意天数）
WITH FirstActivity AS (
    SELECT user_id,
           MIN(activity_date) as first_date
    FROM UserActivity
    GROUP BY user_id
)
SELECT fa.first_date as cohort_date,
       DATEDIFF(ua.activity_date, fa.first_date) as days_since_first,
       COUNT(DISTINCT fa.user_id) as cohort_size,
       COUNT(DISTINCT ua.user_id) as retained_users,
       ROUND(COUNT(DISTINCT ua.user_id) * 100.0 /
             COUNT(DISTINCT fa.user_id), 2) as retention_rate
FROM FirstActivity fa
LEFT JOIN UserActivity ua ON fa.user_id = ua.user_id
GROUP BY fa.first_date, DATEDIFF(ua.activity_date, fa.first_date)
ORDER BY cohort_date, days_since_first;
```

**关键点：**
1. 先找到每个用户的首次活动日期
2. 使用 `CASE WHEN` 统计不同天数的留存
3. `DATEDIFF` 计算日期差

### 10. 实现行转列（PIVOT）

**题目：** 将成绩表从行格式转换为列格式。

```sql
-- 表结构
CREATE TABLE Scores (
    student_id INT,
    subject VARCHAR(50),
    score INT
);

-- 示例数据
INSERT INTO Scores VALUES
(1, 'Math', 90),
(1, 'English', 85),
(1, 'Science', 92),
(2, 'Math', 88),
(2, 'English', 90),
(2, 'Science', 85);
```

**实现：**

```sql
-- 方法1: 使用 CASE WHEN
SELECT student_id,
       MAX(CASE WHEN subject = 'Math' THEN score END) as Math,
       MAX(CASE WHEN subject = 'English' THEN score END) as English,
       MAX(CASE WHEN subject = 'Science' THEN score END) as Science,
       AVG(score) as Average
FROM Scores
GROUP BY student_id;

-- 方法2: 使用 IF (MySQL)
SELECT student_id,
       MAX(IF(subject = 'Math', score, NULL)) as Math,
       MAX(IF(subject = 'English', score, NULL)) as English,
       MAX(IF(subject = 'Science', score, NULL)) as Science
FROM Scores
GROUP BY student_id;

-- 动态 PIVOT（需要存储过程）
DELIMITER $$
CREATE PROCEDURE DynamicPivot()
BEGIN
    SET @sql = NULL;

    SELECT GROUP_CONCAT(DISTINCT
        CONCAT('MAX(CASE WHEN subject = ''', subject,
               ''' THEN score END) AS `', subject, '`')
    ) INTO @sql
    FROM Scores;

    SET @sql = CONCAT('SELECT student_id, ', @sql,
                     ', AVG(score) as Average FROM Scores GROUP BY student_id');

    PREPARE stmt FROM @sql;
    EXECUTE stmt;
    DEALLOCATE PREPARE stmt;
END$$
DELIMITER ;

-- 调用
CALL DynamicPivot();

-- 列转行（UNPIVOT）
WITH Pivoted AS (
    SELECT student_id,
           MAX(CASE WHEN subject = 'Math' THEN score END) as Math,
           MAX(CASE WHEN subject = 'English' THEN score END) as English,
           MAX(CASE WHEN subject = 'Science' THEN score END) as Science
    FROM Scores
    GROUP BY student_id
)
SELECT student_id, 'Math' as subject, Math as score FROM Pivoted WHERE Math IS NOT NULL
UNION ALL
SELECT student_id, 'English' as subject, English as score FROM Pivoted WHERE English IS NOT NULL
UNION ALL
SELECT student_id, 'Science' as subject, Science as score FROM Pivoted WHERE Science IS NOT NULL
ORDER BY student_id, subject;
```

**关键点：**
1. `CASE WHEN` + `MAX/MIN` 实现行转列
2. 动态 SQL 处理未知列数
3. `UNION ALL` 实现列转行

### 11. 实现树形查询（递归）

**题目：** 查询组织架构树，找出所有下属。

```sql
-- 表结构
CREATE TABLE Employees (
    id INT PRIMARY KEY,
    name VARCHAR(50),
    manager_id INT
);

-- 示例数据
INSERT INTO Employees VALUES
(1, 'CEO', NULL),
(2, 'VP Sales', 1),
(3, 'VP Tech', 1),
(4, 'Manager A', 2),
(5, 'Manager B', 3),
(6, 'Employee 1', 4),
(7, 'Employee 2', 4);
```

**实现：**

```sql
-- 方法1: 递归 CTE（MySQL 8.0+）
WITH RECURSIVE EmployeeHierarchy AS (
    -- 锚点：找到指定经理
    SELECT id, name, manager_id, 1 as level, CAST(name AS CHAR(200)) as path
    FROM Employees
    WHERE id = 1  -- CEO

    UNION ALL

    -- 递归：找到下属
    SELECT e.id, e.name, e.manager_id, eh.level + 1,
           CONCAT(eh.path, ' -> ', e.name)
    FROM Employees e
    JOIN EmployeeHierarchy eh ON e.manager_id = eh.id
)
SELECT id,
       CONCAT(REPEAT('  ', level - 1), name) as name,
       level,
       path
FROM EmployeeHierarchy
ORDER BY path;

-- 查询某人的所有上级
WITH RECURSIVE ManagerChain AS (
    SELECT id, name, manager_id, 1 as level
    FROM Employees
    WHERE id = 6  -- 某员工

    UNION ALL

    SELECT e.id, e.name, e.manager_id, mc.level + 1
    FROM Employees e
    JOIN ManagerChain mc ON e.id = mc.manager_id
)
SELECT * FROM ManagerChain ORDER BY level DESC;

-- 计算每个节点的下属数量
WITH RECURSIVE EmployeeTree AS (
    SELECT id, name, manager_id
    FROM Employees

    UNION ALL

    SELECT et.id, et.name, et.manager_id
    FROM EmployeeTree et
    JOIN Employees e ON et.manager_id = e.id
)
SELECT e.id,
       e.name,
       COUNT(et.id) - 1 as subordinate_count
FROM Employees e
LEFT JOIN EmployeeTree et ON e.id = et.id OR et.manager_id = e.id
GROUP BY e.id, e.name
ORDER BY subordinate_count DESC;

-- 方法2: 使用路径编码（避免递归）
-- 为每个节点分配路径编码（如：1.2.4）
ALTER TABLE Employees ADD COLUMN path VARCHAR(200);

UPDATE Employees SET path = '1' WHERE id = 1;
UPDATE Employees e1
JOIN Employees e2 ON e1.manager_id = e2.id
SET e1.path = CONCAT(e2.path, '.', e1.id);

-- 查询所有下属（路径匹配）
SELECT *
FROM Employees
WHERE path LIKE (
    SELECT CONCAT(path, '%')
    FROM Employees
    WHERE id = 1
);
```

**关键点：**
1. `WITH RECURSIVE` 实现递归查询
2. 锚点查询 + 递归部分
3. 路径编码避免递归，提高性能

### 12. 实现数据去重和合并

**题目：** 合并重复的客户记录，保留最完整的信息。

```sql
-- 表结构
CREATE TABLE Customers (
    id INT PRIMARY KEY AUTO_INCREMENT,
    email VARCHAR(100),
    phone VARCHAR(20),
    name VARCHAR(100),
    address VARCHAR(200),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 示例数据（有重复）
INSERT INTO Customers (email, phone, name, address) VALUES
('john@example.com', '1234567890', 'John Doe', '123 Main St'),
('john@example.com', NULL, 'John', NULL),
(NULL, '1234567890', 'John D.', '123 Main Street'),
('jane@example.com', '0987654321', 'Jane Smith', '456 Oak Ave');
```

**实现：**

```sql
-- 步骤1: 识别重复组
WITH DuplicateGroups AS (
    SELECT *,
           -- 使用 email 或 phone 分组
           COALESCE(email, phone) as group_key,
           ROW_NUMBER() OVER (
               PARTITION BY COALESCE(email, phone)
               ORDER BY
                   -- 优先选择信息最完整的记录
                   (CASE WHEN email IS NOT NULL THEN 1 ELSE 0 END +
                    CASE WHEN phone IS NOT NULL THEN 1 ELSE 0 END +
                    CASE WHEN name IS NOT NULL THEN 1 ELSE 0 END +
                    CASE WHEN address IS NOT NULL THEN 1 ELSE 0 END) DESC,
                   created_at DESC
           ) as rn
    FROM Customers
)
SELECT * FROM DuplicateGroups WHERE rn = 1;

-- 步骤2: 合并信息（取最佳值）
CREATE TABLE MergedCustomers AS
WITH DuplicateGroups AS (
    SELECT *,
           COALESCE(email, phone) as group_key
    FROM Customers
)
SELECT MIN(id) as id,
       MAX(email) as email,
       MAX(phone) as phone,
       MAX(name) as name,
       MAX(address) as address,
       MIN(created_at) as created_at,
       COUNT(*) as merged_count
FROM DuplicateGroups
GROUP BY group_key;

-- 步骤3: 使用 COALESCE 合并字段
WITH RankedCustomers AS (
    SELECT *,
           ROW_NUMBER() OVER (
               PARTITION BY COALESCE(email, phone)
               ORDER BY created_at
           ) as rn
    FROM Customers
)
SELECT COALESCE(c1.email, c2.email, c3.email) as email,
       COALESCE(c1.phone, c2.phone, c3.phone) as phone,
       COALESCE(c1.name, c2.name, c3.name) as name,
       COALESCE(c1.address, c2.address, c3.address) as address
FROM RankedCustomers c1
LEFT JOIN RankedCustomers c2
    ON c1.group_key = c2.group_key AND c2.rn = 2
LEFT JOIN RankedCustomers c3
    ON c1.group_key = c3.group_key AND c3.rn = 3
WHERE c1.rn = 1;

-- 高级: 使用模糊匹配（Levenshtein 距离）
-- 需要安装 UDF 或使用存储过程
DELIMITER $$
CREATE FUNCTION LevenshteinDistance(s1 VARCHAR(255), s2 VARCHAR(255))
RETURNS INT
DETERMINISTIC
BEGIN
    DECLARE len1, len2, i, j, cost INT;
    DECLARE d VARBINARY(65535);

    SET len1 = CHAR_LENGTH(s1);
    SET len2 = CHAR_LENGTH(s2);
    SET d = '';

    -- 初始化
    SET i = 0;
    WHILE i <= len1 DO
        SET d = CONCAT(d, UNHEX(HEX(i)));
        SET i = i + 1;
    END WHILE;

    -- 计算距离（简化版）
    -- 实际实现需要完整的动态规划算法

    RETURN ABS(len1 - len2);  -- 简化返回
END$$
DELIMITER ;
```

**关键点：**
1. 使用 `COALESCE` 选择第一个非 NULL 值
2. `ROW_NUMBER()` 选择最佳记录
3. 模糊匹配需要自定义函数

### 13. 设计秒杀库存扣减的 SQL

**需求：** 高并发场景下的库存扣减，防止超卖。

```sql
-- 表结构
CREATE TABLE Products (
    id INT PRIMARY KEY,
    name VARCHAR(100),
    stock INT,
    version INT DEFAULT 0,  -- 乐观锁版本号
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE Orders (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    product_id INT,
    quantity INT,
    status VARCHAR(20),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 示例数据
INSERT INTO Products VALUES (1, 'iPhone 15', 100, 0, NOW());
```

**实现：**

```sql
-- 方法1: 悲观锁（FOR UPDATE）
START TRANSACTION;

SELECT stock FROM Products WHERE id = 1 FOR UPDATE;

-- 检查库存
UPDATE Products
SET stock = stock - 1
WHERE id = 1 AND stock > 0;

-- 如果 affected rows = 1，则创建订单
INSERT INTO Orders (user_id, product_id, quantity, status)
VALUES (123, 1, 1, 'pending');

COMMIT;

-- 方法2: 乐观锁（版本号）
-- 读取当前版本
SELECT stock, version FROM Products WHERE id = 1;

-- 更新时检查版本号
UPDATE Products
SET stock = stock - 1,
    version = version + 1
WHERE id = 1
  AND stock > 0
  AND version = @old_version;  -- 从上面查询得到

-- 检查 affected rows，如果为 0 则重试

-- 方法3: 直接扣减（最简单，依赖数据库约束）
UPDATE Products
SET stock = stock - 1
WHERE id = 1 AND stock > 0;

-- 如果 affected rows = 1，则成功
IF ROW_COUNT() > 0 THEN
    INSERT INTO Orders (user_id, product_id, quantity, status)
    VALUES (123, 1, 1, 'paid');
END IF;

-- 方法4: 使用队列表
CREATE TABLE StockQueue (
    id INT PRIMARY KEY AUTO_INCREMENT,
    product_id INT,
    user_id INT,
    status VARCHAR(20),  -- pending, processing, completed, failed
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 先入队
INSERT INTO StockQueue (product_id, user_id, status)
VALUES (1, 123, 'pending');

-- 后台任务处理队列
-- 1. 获取待处理记录
-- 2. 扣减库存
-- 3. 创建订单
-- 4. 更新队列状态

-- 存储过程示例
DELIMITER $$
CREATE PROCEDURE ProcessSecKill(
    IN p_user_id INT,
    IN p_product_id INT,
    IN p_quantity INT
)
BEGIN
    DECLARE v_stock INT;
    DECLARE v_affected INT;
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SELECT 'Error occurred' as result;
    END;

    START TRANSACTION;

    -- 锁定库存行
    SELECT stock INTO v_stock
    FROM Products
    WHERE id = p_product_id
    FOR UPDATE;

    -- 检查库存
    IF v_stock >= p_quantity THEN
        -- 扣减库存
        UPDATE Products
        SET stock = stock - p_quantity
        WHERE id = p_product_id;

        -- 创建订单
        INSERT INTO Orders (user_id, product_id, quantity, status)
        VALUES (p_user_id, p_product_id, p_quantity, 'paid');

        COMMIT;
        SELECT 'Success' as result, LAST_INSERT_ID() as order_id;
    ELSE
        ROLLBACK;
        SELECT 'Insufficient stock' as result;
    END IF;
END$$
DELIMITER ;

-- 调用
CALL ProcessSecKill(123, 1, 1);
```

**关键点：**
1. 悲观锁适合冲突高的场景
2. 乐观锁适合冲突低的场景
3. 队列削峰，异步处理
4. 使用存储过程减少网络往返

### 14. 设计数据分片查询

**需求：** 按时间范围分片的日志表，实现高效查询。

```sql
-- 分表策略：按月分表
CREATE TABLE logs_202401 (
    id BIGINT PRIMARY KEY,
    user_id INT,
    action VARCHAR(50),
    created_at TIMESTAMP,
    INDEX idx_user_time (user_id, created_at)
) PARTITION BY RANGE (TO_DAYS(created_at)) (
    PARTITION p0 VALUES LESS THAN (TO_DAYS('2024-01-01')),
    PARTITION p1 VALUES LESS THAN (TO_DAYS('2024-02-01')),
    PARTITION p2 VALUES LESS THAN (TO_DAYS('2024-03-01')),
    PARTITION p3 VALUES LESS THAN (MAXVALUE)
);

-- 跨表查询（使用视图）
CREATE VIEW logs_all AS
SELECT * FROM logs_202401
UNION ALL
SELECT * FROM logs_202402
UNION ALL
SELECT * FROM logs_202403;

-- 动态查询存储过程
DELIMITER $$
CREATE PROCEDURE QueryLogsByDateRange(
    IN p_start_date DATE,
    IN p_end_date DATE,
    IN p_user_id INT
)
BEGIN
    DECLARE v_table VARCHAR(20);
    DECLARE v_sql TEXT;
    DECLARE done INT DEFAULT FALSE;
    DECLARE cur CURSOR FOR
        SELECT CONCAT('logs_', DATE_FORMAT(p_start_date, '%Y%m'));

    -- 构建动态 SQL
    SET @sql = '';
    SET @current_date = p_start_date;

    WHILE @current_date <= p_end_date DO
        SET v_table = CONCAT('logs_', DATE_FORMAT(@current_date, '%Y%m'));

        IF @sql != '' THEN
            SET @sql = CONCAT(@sql, ' UNION ALL ');
        END IF;

        SET @sql = CONCAT(@sql,
            'SELECT * FROM ', v_table,
            ' WHERE user_id = ', p_user_id,
            ' AND created_at BETWEEN ''', p_start_date, ''' AND ''', p_end_date, '''');

        SET @current_date = DATE_ADD(@current_date, INTERVAL 1 MONTH);
    END WHILE;

    PREPARE stmt FROM @sql;
    EXECUTE stmt;
    DEALLOCATE PREPARE stmt;
END$$
DELIMITER ;

-- 调用
CALL QueryLogsByDateRange('2024-01-01', '2024-03-31', 123);

-- 使用分区表（推荐）
CREATE TABLE logs (
    id BIGINT AUTO_INCREMENT,
    user_id INT,
    action VARCHAR(50),
    created_at TIMESTAMP,
    PRIMARY KEY (id, created_at),
    INDEX idx_user (user_id)
) PARTITION BY RANGE (YEAR(created_at) * 100 + MONTH(created_at)) (
    PARTITION p202401 VALUES LESS THAN (202402),
    PARTITION p202402 VALUES LESS THAN (202403),
    PARTITION p202403 VALUES LESS THAN (202404),
    PARTITION pmax VALUES LESS THAN MAXVALUE
);

-- 分区裁剪（自动优化）
SELECT * FROM logs
WHERE created_at BETWEEN '2024-01-01' AND '2024-01-31'
  AND user_id = 123;

-- 查看分区信息
SELECT PARTITION_NAME, TABLE_ROWS
FROM INFORMATION_SCHEMA.PARTITIONS
WHERE TABLE_NAME = 'logs';

-- 添加新分区
ALTER TABLE logs
ADD PARTITION (PARTITION p202404 VALUES LESS THAN (202405));

-- 删除旧分区
ALTER TABLE logs DROP PARTITION p202401;
```

**关键点：**
1. 分区键要在 WHERE 条件中
2. 分区裁剪提高查询性能
3. 定期维护分区（添加/删除）
4. 使用分区表优于手动分表
